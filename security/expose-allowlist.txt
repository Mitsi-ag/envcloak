# Files allowed to call secrecy::ExposeSecret::expose_secret or
# ExposeSecretMut::expose_secret_mut, which clippy.toml otherwise forbids.
#
# One repo-relative path per line, followed by `#` and the reason. Every
# call site still needs `#[allow(clippy::disallowed_methods)]`, and
# scripts/check-unsafe.sh fails on such an allow in any file not listed here,
# on the names expose_secret, expose_secret_mut, ExposeSecret and
# ExposeSecretMut in any file not listed here (in any cfg, which clippy does
# not lint for targets CI does not build), on a macro defined in a listed
# file, and on entries whose file no longer exists. Adding a line needs
# review: each one is a place a plaintext value can escape.

crates/envcloak-core/src/secret.rs           # SecretBytes and SecretBuf themselves, and their unit tests
crates/envcloak-core/src/crypto/aead.rs      # seal and open: read a subkey to encrypt or decrypt
crates/envcloak-core/src/crypto/envelope.rs  # wraps and unwraps the VMK under keys derived from a KEK
crates/envcloak-core/src/crypto/kdf.rs       # Argon2id reads the passphrase or Recovery Kit
crates/envcloak-core/src/crypto/keys.rs      # derives subkeys from the VMK; keyed hashing under a subkey; compares two VMKs in constant time; VMK export for tests (feature testing only)
crates/envcloak-core/src/vault/values.rs     # the vault's plaintext boundary: seals, opens and hashes values and prior values, re-seals them in a migration, hands opened metadata to its decoder
crates/envcloak-core/src/passphrase.rs       # reads a new passphrase to check its length and the common-password list
crates/envcloak-core/src/recovery.rs         # encodes the Recovery Kit for display and decodes a typed kit
crates/envcloak-policy/src/envfile.rs        # parses an --env-file in place; copies each value only into a SecretBuf
crates/envcloak-scan/src/dotenv.rs           # parses a dotenv file in place for import; copies each value only into a SecretBuf
crates/envcloak-scan/src/restore.rs          # writes an env file: back from its encrypted backup for init --undo, after a proof; or with the entries the vault holds taken out, for init --delete-plaintext; and a file back from a backup v2's chunks, after a proof, only while it is what the change left
crates/envcloak-providers/src/detect.rs      # matches a value against the registry's key patterns in place; returns provider ids and a classification, never the value
crates/envcloak-ipc/src/frame.rs             # frame bodies: parsed in place and written to a verified peer
crates/envcloak-ipc/src/wire_secret.rs       # base64-encodes a value into a request or response
crates/envcloak-daemon/src/redact.rs          # hands a request's values to the redactor that masks the command line its audit entry keeps
crates/envcloak-exec/src/coverage.rs         # hands a run's values to the redactor that masks the command's output
crates/envcloak-exec/src/spawn.rs            # puts a run's values in the child's environment, and nowhere else
crates/envcloak-agents/src/hook/payload.rs   # the hook handler reads a host's payload in place to decide; only the decision and fixed text leave it
crates/envcloak-signin/src/totp.rs           # RFC 6238 reads a seed into HMAC; generated digits stay in SecretBytes, never formatting or I/O
crates/envcloak-signin/src/otpauth.rs        # enrollment parsed in place and decoded into wiping buffers; only typed parameters and SecretBytes leave

crates/envcloak-scan/src/profile.rs          # parses bounded profile literals in place; never executes a profile
crates/envcloak-scan/src/candidates.rs       # per-run keyed candidate hashes, never persisted or reported
crates/envcloak-scan/src/agent_config.rs     # config bindings and envFile paths read only inside approved roots
crates/envcloak-scan/src/json.rs             # bounded duplicate-key fingerprints; string decoding stays in wiping buffers
